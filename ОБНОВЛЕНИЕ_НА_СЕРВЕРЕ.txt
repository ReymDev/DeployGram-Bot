# 1. ПОДКЛЮЧИТЬСЯ К СЕРВЕРУ:
ssh root@94.183.254.95

# 2. ОБНОВИТЬ ГЛАВНЫЙ БОТ:
cd ~/DeployGram-Bot
git pull
cd docker
docker compose -p mainbot down
docker compose -p mainbot up -d --build

# 3. ОБНОВИТЬ LOCATE AGENT:
cd ~/DeployGram-locate
git pull
cd docker
docker compose -p locate down
docker compose -p locate up -d --build

# 4. ИСПРАВИТЬ ПРАВА НА СУЩЕСТВУЮЩИЕ ПРОЕКТЫ (если есть):
chown -R 1000:1000 /var/lib/deploygram/data/projects/

# 5. ПРОВЕРИТЬ СТАТУС:
docker ps --format "table {{.Names}}\t{{.Status}}\t{{.Ports}}"

# Должно быть:
# mainbot-bot-1     Up X seconds
# mainbot-agent-1   Up X seconds   0.0.0.0:8080->8080/tcp
# mainbot-redis-1   Up X seconds   0.0.0.0:6379->6379/tcp
# locate-agent-1    Up X seconds   0.0.0.0:8081->8080/tcp

# 6. ПРОВЕРИТЬ ЛОГИ:
docker logs mainbot-bot-1 --tail 20
docker logs locate-agent-1 --tail 20

# 7. HEALTH CHECK:
curl http://localhost:8080/health
curl http://localhost:8081/health
